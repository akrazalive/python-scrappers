  39  def _has_monetary_price(store_prices: dict, expected_modality: str = "IN_STORE") -> bool:
  40      """Return whether a store-price block contains a payable value for the selected modality."""
  41      for kind in ("regular", "promo"):
  42          value = store_prices.get(kind)
  43          if not isinstance(value, dict):
  44              continue
  45          mode = str(value.get("modality") or value.get("fulfillmentMode") or value.get("fulfillmentType") or "").upper()
  46          if mode and mode != expected_modality.upper():
  47              continue
  48          for key in ("price", "amount", "nforPrice"):
  49              if _positive_number(value.get(key)):
  50                  return True
  51          text = value.get("defaultDescription")
  52          if isinstance(text, str):
  53              match = _DISPLAY_PRICE_RE.fullmatch(text)
  54              if match and float(match.group(1)) > 0:
  55                  return True
  56      return False

 317      async def crawl_pending(self) -> None:
 318          """Continuous work queue: every pending request starts as soon as the fetcher has a free slot, and requests
 319          queued by expansion / brand slices start as soon as they appear. (Up to 1.1.1 the crawl ran in waves: a wave
 320          waited for its slowest request, retries and back-off included, before the next wave started, leaving most
 321          parallel slots idle - the main reason for ~8-9 requests per minute on 21 Oct.) The fetcher's semaphore still
 322          bounds parallel requests (`fetch.concurrency`). Each request is started at most once per session; one left
 323          pending is picked up by the next resume."""
 324          running: dict[tuple, asyncio.Task] = {}
 325          started: set[tuple] = set()
 326          n = 0
 327          while True:
 328              for l in self.ck.pending():
 329                  key = (l["category"], l["url"])
 330                  if key not in started:
 331                      started.add(key)
 332                      running[key] = asyncio.create_task(self.crawl_listing(l))
 333              if not running:
 334                  break
 335              done, _ = await asyncio.wait(running.values(), return_when=asyncio.FIRST_COMPLETED)
 336              for key in [k for k, t in running.items() if t in done]:
 337                  exc = running.pop(key).exception()
 338                  if exc:
 339                      for t in running.values():
 340                          t.cancel()
 341                      raise exc
 342              n += len(done)
 343              if n % 50 < len(done):
 344                  log.info("crawl", extra={"requests_done": n, "requests_running": len(running),
 345                                           "requests_total": len(self.ck.listings()), "fetch": dict(self.fetcher.stats)})

 347      async def crawl_listing(self, l: dict) -> None:
 348          """Read one search request of a shelf.
 349  
 350          Mariano's server-rendered search returns the first 24 results of a request whatever page is asked for.
 351          Two mechanisms extend coverage: (1) brand slices – a request that reports more products than it returned is
 352          partitioned with Mariano's own `brandName` filter, one request per brand (slices are disjoint, and their
 353          reported totals reconcile against the parent request in coverage.json); (2) query expansion – new products
 354          queue queries built from their sub-commodity names and brands. Neither is capped by default."""
 355          cat, url, sub = l["category"], l["url"], l["sub_category"]
 356          page, streak = int(l["next_page"]), int(l["empty_streak"])
 357          paginate = bool(self.s.get("crawl.paginate", False))
 358          mode = l.get("mode") or "query"
 359          while True:
 360              if self.limit and self.ck.kept_in_category(cat) >= self.limit:
 361                  self.ck.finish_listing(cat, url, "demo_limit_reached")
 362                  return
 363              where = {"category": cat, "sub": sub, "query": l["query"]}
 364              try:
 365                  sp, _, sha = await self._search(_with_page(url, page), where, kind=f"search_{mode}")
 366              except FetchError as exc:
 367                  self.ck.fail_listing(cat, url, f"page {page}: {exc}")
 368                  self.ck.event("listing_failed", {"category": cat, "sub": sub, "query": l["query"], "error": str(exc)})
 369                  return
 370              if not self._store_ok(sp.store, f"{cat} / {sub} [{l['query']}]") or (
 371                      sp.request_location and sp.request_location != self.loc.store_id):
 372                  self.ck.fail_listing(cat, url, f"page {page}: store context mismatch")
 373                  return
 374              if mode == "brand_slice" and sp.upcs:
 375                  want = l["query"].split("| brand=", 1)[-1].strip().lower()
 376                  got = [((sp.products.get(u) or {}).get("data") or {}).get("item", {}).get("brand", {}).get("name", "").lower()
 377                         for u in sp.upcs]
 378                  share = sum(1 for b in got if b == want) / len(got)
 379                  if share < float(self.s.get("crawl.brand_slice_min_share", 0.9)):
 380                      # Mariano's ignored the brand filter: never count this response as coverage of the slice
 381                      self.ck.fail_listing(cat, url, f"brand filter not honored ({share:.0%} of products are '{want}')")
 382                      self.ck.event("brand_filter_not_honored", {"query": l["query"], "share": share})
 383                      return
 384              accepted, skipped, matched, terms, obs, excl, listed = [], [], 0, [], [], [], []
 385              coupon_rows: list[tuple[str, str, str]] = []

 446      def _slice(self, l: dict, sp: parse.SearchPage, listed: list[tuple[str, str]]) -> int:
 447          """Partition a SEED request (the shelf's own name / parts) that reports more than it returned into brand slices
 448          (Mariano's brandName filter). Expansion requests are discovery helpers and are not partitioned."""
 449          if not self.s.get("crawl.brand_slices", True) or sp.total <= len(sp.upcs):
 450              return 0
 451          brands = {b for _, b in listed if b}
 452          # brands seen anywhere in this shelf can also hide in this request's unseen remainder
 453          for (b,) in self.ck.db.execute(
 454                  "SELECT DISTINCT li.brand FROM listing_items li JOIN listings l ON l.url = li.url "
 455                  "WHERE l.category=? AND l.sub_category=? AND li.brand != ''", (l["category"], l["sub_category"])):
 456              brands.add(b)
 457          cap = int(self.s.get("crawl.max_brand_slices_per_request", 0))
 458          added = 0
 459          for b in sorted(brands):
 460              if cap and added >= cap:
 461                  break
 462              sep = "&" if "?" in l["url"] else "?"
 463              if self.ck.add_listing(l["category"], l["sub_category"], f"{l['url']}{sep}brandName={quote(b)}",
 464                                     f"{l['query']} | brand={b}", l["taxonomy"], bool(l["curated"]),
 465                                     mode="brand_slice", parent=l["url"]):
 466                  added += 1
 467          return added

 469      def _expand(self, l: dict, terms: list[str]) -> int:
 470          cap = int(self.s.get("crawl.max_queries_per_shelf", 0))
 471          shelf = [x for x in self.ck.listings() if x["category"] == l["category"] and x["sub_category"] == l["sub_category"]
 472                   and (x.get("mode") or "query") in ("query", "expansion")]
 473          used = {x["query"].lower() for x in shelf}
 474          added = 0
 475          for t in terms:
 476              if cap and len(shelf) + added >= cap:
 477                  break
 478              if t.lower() in used:
 479                  continue
 480              used.add(t.lower())
 481              self.ck.add_listing(l["category"], l["sub_category"],
 482                                  f"{self.base}/search?query={quote(t)}&searchType=default_search",
 483                                  t, l["taxonomy"], bool(l["curated"]), mode="expansion")
 484              added += 1
 485          return added

 105      async def get(self, url: str, render_js: bool = False, validate=None, retries: int | None = None) -> FetchResult:
 106          """Fetch with retries. `validate(html)` may raise to mark a response unusable (retried like a block)."""
 107          last: Exception | None = None
 108          max_retries = self.retries if retries is None else retries
 109          for attempt in range(max_retries + 1):
 110              self.validate_request(url, render_js)
 111              if self.budget and self.stats["requests"] >= self.budget:
 112                  raise FetchError(f"request budget of {self.budget} reached", "budget")
 113              async with self.sem:
 114                  await self._pace()
 115                  from . import lock
 116                  try:
 117                      lock.check_active()                 # a lost lease stops every further request
 118                  except lock.LockLost as exc:
 119                      raise FetchError(str(exc), "lock") from exc
 120                  if self.total_budget is not None:   # counted before sending; raises once the test's total is used
 121                      self.total_budget.take(url, self.loc.store_id)
 122                  self.stats["requests"] += 1
 123                  try:
 124                      res = await asyncio.wait_for(self._fetch_once(url, render_js), self.timeout + 90)
 125                      self.stats["seconds"] += res.elapsed
 126                      if res.elapsed > self.slow:
 127                          self.stats["slow"] += 1
 128                          log.warning("slow response", extra={"url": url, "seconds": round(res.elapsed, 1)})

 131                              exc = FetchError(f"target refused HTTP {res.status}", "refused")
 132                              exc.result = res
 133                              if self.on_refusal:

 135                              raise exc
 136                          raise FetchError(f"HTTP {res.status}", "http")

 143                  except FetchError as exc:

   2  
   3  The per-process caps (`--max-requests`) restart with each process; this budget does not. It is a pair of files in the
   4  state directory (Azure: the mounted Azure Files share), so it holds across commands, retries, resumes, container
   5  restarts and repeated job executions:
   6  
   7      state/_budgets/<name>.json     {"name", "total", "used", "created_at", "updated_at"}
   8      state/_budgets/<name>.jsonl    one line per provider request: ts, n, command, store, url
   9  
  10  Exclusive ownership (1.1.5): a budget may only be opened and used while this process holds the live lock
  11  (lock.live_lock – a Blob lease in Azure), so two executions can never draw on it at the same time. Every take()
  12  re-checks the lock, re-reads the state from disk (no cached counter) and only then reserves the next request.
  13  
  14  Order of writes per request: (1) the ledger line is appended and flushed, (2) the state file is replaced atomically,
  15  (3) the request is sent. A crash between (1) and (2) leaves one ledger line more than `used`; on the next open that
  16  request is counted as used (fail safe). Anything else is inconsistent and fails closed (BudgetError): a missing or
  17  invalid state file next to an existing ledger, fewer ledger lines than `used`, more than one extra line, a total that
  18  differs from the requested one, or used > total. A cap can only be raised by a new, separately named budget.

  83              raise BudgetError(f"budget {self.name!r}: used {used} outside 0..{total}")
  84          lines = self._lines()
  85          if lines == used + 1 and lines <= total:      # ledger written, state not yet: count it as used
  86              st["used"], st["updated_at"], st["repaired_at"] = lines, _now(), _now()
  87              self.state = st
  88              self._save()
  89          elif lines != used:
  90              raise BudgetError(f"budget {self.name!r}: ledger has {lines} lines but state says used={used}")
  91          return st
  92  
  93      @property
  94      def total(self) -> int:
  95          return int(self.state["total"])
  96  
  97      @property
  98      def used(self) -> int:
  99          return int(self.state["used"])
 100  
 101      @property
 102      def remaining(self) -> int:
 103          return self.total - self.used
 104  
 105      def _save(self):
 106          tmp = self.path.with_suffix(".tmp")
 107          tmp.write_text(json.dumps(self.state, indent=2), encoding="utf-8")
 108          os.replace(tmp, self.path)
 109  
 110      def take(self, url: str, store_id: str = "") -> int:
 111          """Reserve one provider request before it is sent; refuse when the total is used up or ownership is gone."""
 112          self._owned()
 113          self.state = self._load()                      # never trust an in-memory counter
 114          if self.used >= self.total:
 115              raise FetchError(f"total budget {self.name!r} of {self.total} requests used up", "budget")
 116          n = self.used + 1
 117          with self.log.open("a", encoding="utf-8") as f:   # (1) ledger first
 118              f.write(json.dumps({"ts": _now(), "n": n, "command": self.command, "store": store_id, "url": url}) + "\n")
 119              f.flush()
 120              os.fsync(f.fileno())
 121          self.state = {**self.state, "used": n, "updated_at": _now()}
 122          self._save()                                       # (2) then the counter
 123          return n
 124  
 125      def summary(self) -> dict:
 126          return {**self.state, "remaining": self.remaining, "ledger": str(self.log)}
 127  
 128  
 129  def parse_spec(spec: str) -> tuple[str, int]:
 130      """'test1:19' -> ('test1', 19)"""
 131      name, _, total = (spec or "").partition(":")
 132      if not name or not total.isdigit():
 133          raise ValueError(f"budget must look like NAME:TOTAL (e.g. test1:19), got {spec!r}")
 134      return name.strip(), int(total)
