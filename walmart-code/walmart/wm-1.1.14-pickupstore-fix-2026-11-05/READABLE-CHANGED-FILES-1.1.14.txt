  326      data = next_data(html, final_url)
  327      initial = dig(data, "props", "pageProps", "initialData") or {}

  351  def _is_secondary_stack(stack: dict) -> bool:
  352      """Walmart appends a secondary stack (sub_type puShipFul: not available at this store, ships instead)."""
  353      meta = stack.get("meta") or {}
  354      if str(meta.get("fulfillmentIntent", "")).lower() == "shipping":
  355          return True
  356      try:
  357          interp = (json.loads(meta.get("beacon") or "{}").get("interpretations") or [{}])[0]
  358      except (ValueError, TypeError, AttributeError):
  359          return False
  360      return interp.get("type") == "secondary_stack"

  376  _BROWSE_HREF = re.compile(r'href="(/browse/[^"?#]+/(\d+(?:_\d+)+))[^"]*"[^>]*>(.*?)</a>', re.S)
  377  
  378  
  379  def discover_links(data: Any, html: str = "") -> list[tuple[str, str]]:
  380      """Collect (name, url) pairs for /browse/ category links on a category or listing page."""
  381      seen, out = set(), []
  382      for node in walk(data):
  383          if not isinstance(node, dict):
  384              continue
  385          url = node.get("url")
  386          if not isinstance(url, str) and isinstance(node.get("clickThrough"), dict):
  387              url = node["clickThrough"].get("value")
  388          name =node.get("name") or node.get("title") or node.get("text") or node.get("displayName")
  389          if isinstance(url, str) and "/browse/" in url and isinstance(name, str):
  390              path = url.split("?")[0]
  391              if path not in seen:
  392                  seen.add(path)
  393                  out.append((clean_text(name), urljoin(BASE, path)))
  394      for href, _id, label in _BROWSE_HREF.findall(html or ""):
  395          if href not in seen:
  396              seen.add(href)
  397              out.append((clean_text(re.sub("<[^>]+>", " ", label)), urljoin(BASE, href)))
  398      return out

  438  def parse_product(html: str, final_url: str = "") -> ProductDetail:
  439      data = next_data(html, final_url)
  440      product = dig(data, "props", "pageProps", "initialData", "data", "product")
  441      if not isinstance(product, dict) or not product.get("usItemId"):
  442          raise ShapeError("product object missing from product page")
  443      pi = product.get("priceInfo") or {}
  444      specs = dig(data, "props", "pageProps", "initialData", "data", "idml", "specifications") or []
  445      size = ""
  446      ranked = {"net content": 0, "net weight": 1, "size": 2, "capacity": 3, "count per pack": 4, "count": 5}
  447      best = 99
  448      for spec in specs if isinstance(specs, list) else []:
  449          name = clean_text(spec.get("name")).lower() if isinstance(spec, dict) else ""

  455      fulfil = _texts(product.get("fulfillmentOptions") or []) + _texts(product.get("fulfillmentSummary") or [])

  509      address = ", ".join(x for x in (street, city, f"{state} {postal}".strip()) if x)
