  42  def fill(template, **values):
  43      """Recursively replace {placeholders} in a JSON-like provider template."""
  44      if isinstance(template, str):
  45          out = template
  46          for k, v in values.items():
  47              out = out.replace("{" + k + "}", str(v))
  48          return out
  49      if isinstance(template, list):
  50          return [fill(x, **values) for x in template]
  51      if isinstance(template, dict):
  52          return {k: fill(v, **values) for k, v in template.items()}
  53      return template

  83  def search_payload(settings, location, url: str) -> dict:
  84      """`walmart_search` body for a /search URL. Every URL parameter is either carried into the payload through
  85      `fetch.oxylabs.search_params` (URL name -> payload key, e.g. page -> start_page; whole numbers only), consumed (q; stores equal to
  86      the configured store), or the request is REFUSED: a filter or page the route cannot express is never silently
  87      dropped (that would turn page 2 into a repeat of page 1). Category browsing does not use this route."""
  88      u = urlparse(url)
  89      params = parse_qs(u.query, keep_blank_values=True)
  90      mapping = settings.get("fetch.oxylabs.search_params") or {}
  91      query = (params.pop("q", None) or [""])[0].strip()
  92      if not query:
  93          raise FetchError(f"search route: no query in {url}", "compat")
  94      stores = params.pop("stores", None)
  95      if stores and stores != [location.store_id]:
  96          raise FetchError(f"search route: stores={stores} differs from configured store {location.store_id}", "compat")
  97      params.pop("povid", None)  # Walmart click-tracking id, not a filter
  98      extra: dict = {}
  99      for name, values in params.items():
 100          if name not in mapping:
 101              raise FetchError(f"search route: unsupported parameter '{name}' in {url} (refused, not dropped)", "compat")
 102          if len(values) != 1 or not _SEARCH_INT.fullmatch(values[0]):
 103              raise FetchError(f"search route: {name}={values} is not one whole number in {url}", "compat")
 104          value = int(values[0])
 105          if name == "page" and value < 1:
 106              raise FetchError(f"search route: invalid page {values[0]} in {url}", "compat")
 107          extra[mapping[name]] = value
 108      body = fill(settings.get("fetch.oxylabs.search_payload"), url=url, query=query,
 109                  store_id=location.store_id, zipcode=location.zipcode)
 110      return {**body, **extra}

 113  def classic_payload(settings, location, url: str) -> dict:
 114      """Classic Web Scraper API body (realtime.oxylabs.io/v1/queries).
 115      Product pages: the documented `walmart_product` source with store localisation (store_id, delivery_zip,
 116      fulfillment_type) when `fetch.oxylabs.product_payload` is configured; everything else: `fetch.oxylabs.payload`."""
 117      m = _IP_RE.search(url)

 128      name = "base"

 130      def __init__(self, settings: Settings, location: Location):
 131          self.s = settings
 132          self.loc = location
 133          self.timeout = float(settings.get("fetch.timeout_seconds", 90))
 134          self.retries = int(settings.get("fetch.max_retries", 4))
 135          self.backoff = float(settings.get("fetch.backoff_base_seconds", 4))
 136          self.min_delay = float(settings.get("fetch.min_delay_seconds", 2))
 137          self.max_delay = float(settings.get("fetch.max_delay_seconds", 5))
 138          self.slow = float(settings.get("fetch.slow_response_seconds", 45))
 139          self._last = 0.0
 140          self._lock = asyncio.Lock()
 141          self.budget = int(settings.get("fetch.max_requests_per_run", 0))  # 0 = unlimited
 142          self.total_budget = None  # persistent TotalBudget (budget.py), shared by every process of one approved test
 143          self.on_refusal = None
 144          self.stats = {"requests": 0, "retries": 0, "challenges": 0, "http_errors": 0, "slow": 0}
 145  
 146      async def start(self):  # pragma: no cover - overridden where needed
 147          return None
 148  
 149      async def close(self):  # pragma: no cover
 150          return None

 152      async def _fetch_once(self, url: str) -> FetchResult:  # pragma: no cover
 153          raise NotImplementedError

 155      async def _pace(self):
 156          async with self._lock:  # pacing is global even when requests run concurrently
 157              wait = self._last + random.uniform(self.min_delay, self.max_delay) - time.monotonic()
 158              if wait > 0:
 159                  await asyncio.sleep(wait)
 160              self._last = time.monotonic()
 161  
 162      async def get(self, url: str) -> FetchResult:
 163          last_exc: FetchError | None = None
 164          from .. import lock
 165          for attempt in range(1, self.retries + 2):
 166              try:
 167                  lock.check_active()                 # a lost lease stops every further request
 168              except lock.LockLost as exc:
 169                  raise FetchError(str(exc), "lock") from exc
 170              if self.budget and self.stats["requests"] >= self.budget:
 171                  raise FetchError(f"request budget of {self.budget} reached (fetch.max_requests_per_run)", "budget")
 172              await self._pace()
 173              try:
 174                  lock.check_active()             # re-check after pacing, immediately before reserving/sending
 175              except lock.LockLost as exc:
 176                  raise FetchError(str(exc), "lock") from exc
 177              if self.total_budget is not None:  # counted and saved before the request is sent, retries included
 178                  self.total_budget.take(url, self.loc.store_id)
 179              self.stats["requests"] += 1
 180              try:
 181                  res = await self._fetch_once(url)
 182              except FetchError as exc:
 183                  if exc.kind in TERMINAL_KINDS:     # auth / compatibility: retrying cannot help, never retried
 184                      raise
 185                  last_exc = exc
 186              except Exception as exc:  # network, provider errors
 187                  last_exc = FetchError(f"{type(exc).__name__}: {exc}", "error")
 188              else:
 189                  if res.elapsed > self.slow:
 190                      self.stats["slow"] += 1
 191                      log.warning("slow_response", extra={"url": url, "seconds": round(res.elapsed, 1)})
 192                  if res.status >= 400:
 193                      self.stats["http_errors"] += 1
 194                      if res.status in (401, 403):
 195                          if self.on_refusal:
 196                              self.on_refusal(res)
 197                          raise FetchError(f"target refused HTTP {res.status}", "refused")
 198                      last_exc = FetchError(f"HTTP {res.status}", "http")
 199                  elif detect_challenge(res.html, res.final_url):
 200                      self.stats["challenges"] += 1
 201                      last_exc = FetchError("bot challenge / block page", "challenge")
 202                  elif len(res.html) < 2000:
 203                      last_exc = FetchError(f"suspiciously small response ({len(res.html)} bytes)", "error")
 204                  else:
 205                      return res
 206              if attempt <= self.retries:
 207                  self.stats["retries"] += 1
 208                  delay = self.backoff * (2 ** (attempt - 1)) + random.uniform(0, self.backoff)
 209                  log.warning("retry", extra={"url": url, "attempt": attempt, "reason": str(last_exc), "sleep": round(delay, 1)})
 210                  await asyncio.sleep(delay)
 211          raise last_exc or FetchError("unknown fetch failure")

 319      """Plain Chromium session for environments with authorised access.
 320  
 321      No CAPTCHA solving or fingerprint spoofing is performed: if Walmart serves a
 322      challenge, it is detected and reported, and the run fails safely.
 323      """

 325      name = "playwright"
 326  
 327      async def start(self):
 328          from playwright.async_api import async_playwright
 329  
 330          self._pw = await async_playwright().start()
 331          proxy = None
 332          if os.getenv("PLAYWRIGHT_PROXY_SERVER"):
 333              proxy = {"server": os.environ["PLAYWRIGHT_PROXY_SERVER"],
 334                       "username": os.getenv("PLAYWRIGHT_PROXY_USERNAME") or None,
 335                       "password": os.getenv("PLAYWRIGHT_PROXY_PASSWORD") or None}
 336          self._browser = await self._pw.chromium.launch(headless=True, proxy=proxy)

 340              await self._ctx.add_cookies(cookies)

 343      async def close(self):
 344          await self._browser.close()
 345          await self._pw.stop()

 347      async def _fetch_once(self, url: str) -> FetchResult:
 348          t0 = time.monotonic()
 349          resp = await self._page.goto(url, wait_until="domcontentloaded", timeout=self.timeout * 1000)
 350          await self._page.wait_for_timeout(1500)
 351          html = await self._page.content()
 352          return FetchResult(url, self._page.url, resp.status if resp else 0, html, time.monotonic() - t0)

 372  def build_fetcher(settings: Settings, location: Location) -> Fetcher:
 373      name = settings.get("fetch.provider", "oxylabs")
 374      if name not in PROVIDERS:
 375          raise ValueError(f"unknown fetch.provider '{name}', choose one of {sorted(PROVIDERS)}")
 376      fetcher = PROVIDERS[name](settings, location)
 377      spec = settings.get("fetch.total_budget")  # NAME:TOTAL from --budget / WMSCRAPER_BUDGET (cli.py)
 378      if spec:
 379          from ..budget import TotalBudget, parse_spec
 380          from ..storage import build_storage
 381  
 382          bname, total = parse_spec(spec)
 383          fetcher.total_budget = TotalBudget(build_storage(settings), bname, total, command=f"{name} {location.key}")
 384      return fetcher

 337          if self.loc.fulfillment == "pickup":
 338              if (loc.get("intent") or "").upper() != "PICKUP":
 339                  probs.append(f"intent {loc.get('intent') or '-'} (expected PICKUP)")
 340              if ("PICKUP", want) not in [(f.upper(), s) for f, s in p.fulfillment_options]:
 341                  probs.append(f"no PICKUP option for store {want} in fulfillmentSummary {p.fulfillment_options}")
 342          if (p.fulfillment_type or "").upper() != "STORE":
 343              probs.append(f"fulfillmentType {p.fulfillment_type or '-'} (expected STORE: store-sourced offer)")
 344          shown_match = re.fullmatch(r"\s*\$\s*(\d+(?:\.\d{1,2})?)\s*", p.current_price or "")
 345          numeric_ok = isinstance(p.price_value, (int, float)) and not isinstance(p.price_value, bool) \
 346              and math.isfinite(p.price_value) and p.price_value > 0
 347          shown_ok = bool(shown_match and float(shown_match.group(1)) > 0)
 348          if not (numeric_ok or shown_ok):
 349              probs.append("no current price on the store offer")
 350          if self.loc.zipcode and loc.get("postalCode") and loc.get("postalCode") != self.loc.zipcode:
 351              probs.append(f"postalCode {loc.get('postalCode')} != {self.loc.zipcode}")
 352          return probs

 354      async def canary(self, phase: str):
 355          """Every configured canary must pass. Stops at the FIRST failing item (no further request) and raises;
 356          every response is archived with its hash and returned context BEFORE it is validated."""
 357          items = list(self.loc.canary_items)
 358          self.ck.set("canary_configured", bool(items))
 359          if not items:
 360              return
 361          state = self.ck.get("canary", {})
 362          if state.get(phase, {}).get("ok"):
 363              return
 364          checked = []
 365          for item_id in items:
 366              res = await self.fetcher.get(f"https://www.walmart.com/ip/{item_id}")   # terminal errors propagate
 367              arch = self._archive("canary", res, f"canary {phase} {item_id}", always=True)
 368              try:
 369                  probs = self.canary_problems(parse_product(res.html, res.final_url), item_id)
 370              except Exception as exc:
 371                  probs = [f"unparsable product response: {exc}"]
 372              checked.append({"item": item_id, "ok": not probs, "problems": probs,
 373                              "evidence": (arch or {}).get("file"), "sha256": (arch or {}).get("sha256")})
 374              if probs:
 375                  break
 376          ok = all(c["ok"] for c in checked) and len(checked) == len(items)
 377          state[phase] = {"ok": ok, "items": items, "checked": checked,
 378                          "detail": [f"{c['item']}: {'; '.join(c['problems'])}" for c in checked if not c["ok"]],
 379                          "not_requested": items[len(checked):]}
 380          self.ck.set("canary", state)
 381          log.info("canary", extra={"phase": phase, "ok": ok, "checked": checked, "not_requested": items[len(checked):]})
 382          if not ok:
 383              raise FetchError(f"canary failed at {phase}: {state[phase]['detail']} – stopped, "
 384                               f"{len(items) - len(checked)} canary item(s) not requested", "store")

 388          return f"https://www.walmart.com/browse/{cid}"

 390      async def _root_id(self, cat) -> str:
 391          """Real category id for a CSV URL. /browse/ URLs carry it; /cp/ landing pages carry only their own node id
 392          (e.g. /cp/feeding/133283 -> category path 5427_133283), which is resolved from the page's own links."""
 393          cid = cat.category_id
 394          if "/cp/" not in cat.url or "_" in cid:
 395              return cid
 396          try:
 397              res = await self.fetcher.get(cat.url.split("?")[0])
 398              self._archive("category", res, f"{cat.name} landing page")
 399              data = next_data(res.html, res.final_url)
 400              paths = [category_id_of(u) for _n, u in discover_links(data, res.html)]
 401              paths += [k for _n, k in department_children(data)]
 402          except (FetchError, ShapeError, ChallengeError) as exc:
 403              log.warning("root_id_unresolved", extra={"category": cat.name, "error": str(exc)})
 404              return cid
 405          roots = sorted({"_".join(p.split("_")[: p.split("_").index(cid) + 1]) for p in paths if cid in p.split("_")},
 406                         key=len)
 407          root = roots[0] if roots else cid
 408          log.info("category_root", extra={"category": cat.name, "csv_id": cid, "root_id": root})
 409          return root

 412          """Products one listing can expose (Walmart serves at most 25 pages of ~40 products)."""
 413          return int(self.s.get("crawl.max_pages_per_listing", 25)) * int(self.crawl.get("page_size", 40))

 416          """Store-assortment products a listing may hold so that ALL of them (plus online-only items Walmart
 417          interleaves) are reachable within the page window. Above this, the node is split."""
 418          return int(self._window() * float(self.crawl.get("store_window_ratio", 0.75)))

 430      async def discover(self):
 431          """Plan the full store-assortment traversal. Every node is opened store-pinned (no availability facet).
 432          A node whose store assortment exceeds the store window is split into its Departments children, then into
 433          whole-dollar price bands, until every planned listing can be paginated to exhaustion."""
 434          for cat in self.ck.replan_transient_failures():
 435              self.ck.set(f"discovered:{cat}", False)
 436              self.ck.set("discovery_done", False)
 437              log.info("category_replan", extra={"category": cat, "reason": "transient failure during planning"})
 438          if self.ck.get("discovery_done"):
 439              return
 440          sem = asyncio.Semaphore(int(self.crawl.get("category_concurrency", 4)))
 441  
 442          async def one(cat):
 443              async with sem:
 444                  await self._discover_category(cat)
 445  
 446          await asyncio.gather(*(one(c) for c in self.categories))
 447          self.ck.set("discovery_done", True)
 448          self._sync_checkpoint()

 450      def _source_mode(self) -> bool:
 451          """source_count (production): listings are Walmart-sold (retailer facet) and store-pinned; every planned
 452          listing fits the page window and is paginated to Walmart's last page, so the products seen reconcile with
 453          Walmart's own reported total. store_count (legacy): pickup-facet counts as the reconciliation floor."""
 454          return self.crawl.get("coverage", "source_count") == "source_count"

 456      async def _discover_category(self, cat):
 457          if self.ck.get(f"discovered:{cat.name}"):
 458              return
 459          if self._source_mode():
 460              return await self._discover_source(cat)
 461          depth_max = int(self.crawl.get("subcategory_max_depth", 5))
 462          per_cat = int(self.crawl.get("max_listings_per_category", 0))  # 0 = all (production)
 463          skip = [re.compile(p, re.I) for p in self.crawl.get("skip_subcategory_patterns") or []]
 464          excluded = {e.lower() for e in cat.excluded_subcategories}
 465          extra = self._listing_params()
 466          leaves: list[tuple[str, str, int]] = []
 467          root = await self._root_id(cat)
 468          stack = [(cat.name, self._listing_url(root), root, 0)]
 469          seen = set()
 470          while stack and not (per_cat and len(leaves) >= per_cat):
 471              name, url, cid, depth = stack.pop(0)
 472              if cid.split("_")[-1] in seen:
 473                  continue
 474              seen.add(cid.split("_")[-1])
 475              sub = name if depth else cat.name
 476              try:
 477                  res = await self.fetcher.get(with_params(url, **extra))
 478                  lp = parse_listing(res.html, res.final_url)
 479              except ShapeError as exc:
 480                  kids = self._landing_children(res, cid, excluded, skip)
 481                  if kids and depth < depth_max:
 482                      stack[0:0] = [(n, self._listing_url(k), k, depth + 1) for n, k in kids]
 483                      log.info("landing_page_split", extra={"category": cat.name, "node": name, "children": len(kids)})
 484                  elif depth:
 485                      # merchandising/brand landing page with no listing and no child categories: its products are
 486                      # listed under the real departments; recorded, not a coverage gap
 487                      self.ck.event("landing_page_without_listing", {"category": cat.name, "node": name, "url": url})
 488                      log.info("landing_page_without_listing", extra={"category": cat.name, "node": name, "url": url})
 489                  else:
 490                      self.ck.add_listing(cat.name, sub, url)
 491                      self.ck.fail_listing(cat.name, url, f"discovery failed: {exc}")
 492                  continue
 493              except (FetchError, ChallengeError) as exc:
 494                  self.ck.add_listing(cat.name, sub, url)
 495                  self.ck.fail_listing(cat.name, url, f"discovery failed: {exc}")
 496                  log.error("discovery_failed", extra={"category": cat.name, "url": url, "error": str(exc)})
 497                  continue
 498              store_total = await self._store_count(url)
 499              if store_total is None:
 500                  self.ck.add_listing(cat.name, sub, url)
 501                  self.ck.fail_listing(cat.name, url, "store assortment count unavailable")
 502                  continue
 503              if lp.total_count <= self._window() or store_total <= self._store_window():
 504                  if self.loc.only_subcategories and sub.lower() not in {x.lower() for x in self.loc.only_subcategories}:
 505                      continue
 506                  leaves.append((sub, url, store_total))
 507                  self.ck.add_listing(cat.name, sub, url, store_total)
 508                  if not self._check_context(lp.location, url):
 509                      self.ck.fail_listing(cat.name, url, f"page 1: selected store context {lp.location}")
 510                  elif self.ck.listing_row(cat.name, url)["next_page"] == 1:
 511                      self._process_page(cat.name, sub, url, 1, lp, res, int(self.crawl.get("max_pages_per_listing", 25)))
 512                  continue
 513              kids = [(n, k) for n, k in lp.children if k.startswith(cid + "_")
 514                      and n.lower() not in excluded and not any(p.search(n) for p in skip)]
 515              ids = {k for _n, k in kids}
 516              kids = [(n, k) for n, k in kids if not any(k != o and k.startswith(o + "_") for o in ids)]
 517              if kids and depth < depth_max:
 518                  stack[0:0] = [(n, self._listing_url(k), k, depth + 1) for n, k in kids]
 519                  log.info("category_split", extra={"category": cat.name, "node": name, "walmart_total": lp.total_count,
 520                                                    "store_assortment": store_total, "children": len(kids)})
 521              else:
 522                  leaves += await self._price_partitions(cat.name, sub, url)
 523          if self.loc.only_subcategories:
 524              keep = {x.lower() for x in self.loc.only_subcategories}
 525              leaves = [l for l in leaves if l[0].lower() in keep]
 526          for sub, url, expected in leaves:
 527              self.ck.add_listing(cat.name, sub, url, expected)
 528          self.ck.set(f"discovered:{cat.name}", True)
 529          log.info("category_discovered", extra={"category": cat.name, "listings": len(leaves),
 530                                                 "store_assortment_expected": sum(l[2] for l in leaves)})
 531  
 532      async def _discover_source(self, cat):
 533          """Source-count planning. Each node is opened once (store-pinned, Walmart-sold). A node whose Walmart total
 534          fits the page window is a listing (its page 1 is processed at once); a larger node is split into its direct
 535          Departments children, or - when it has none, or fans out into many overlapping merchandising shelves - into
 536          whole-dollar price bands. A node reached again by another path (alias) is planned once. Every planning
 537          decision is checkpointed per node, so a resumed run never re-plans (or re-pays for) finished nodes."""
 538          depth_max = int(self.crawl.get("subcategory_max_depth", 5))
 539          skip = [re.compile(p, re.I) for p in self.crawl.get("skip_subcategory_patterns") or []]
 540          excluded = {e.lower() for e in cat.excluded_subcategories}
 541          root = await self._root_id(cat)
 542          stack = [(cat.name, self._listing_url(root), root, 0)]
 543          seen: dict[str, str] = {}
 544          active = 0
 545          cond = asyncio.Condition()
 546  
 547          async def visit(name, url, cid, depth):
 548              sub = name if depth else cat.name
 549              plan = self.ck.get(f"plan:{cat.name}:{cid}")
 550              if plan is None:
 551                  plan = await self._plan_node(cat.name, sub, url, cid, depth, depth_max, excluded, skip)
 552                  if plan is None:
 553                      return []  # recorded as a failed listing; re-planned on resume when transient
 554                  self.ck.set(f"plan:{cat.name}:{cid}", plan)
 555              return [(n, self._listing_url(k), k, depth + 1) for n, k in plan.get("kids") or []]
 556  
 557          async def worker():  # several nodes of one category are planned at once (shared queue, depth-first order)
 558              nonlocal active
 559              while True:
 560                  async with cond:
 561                      while not stack and active:
 562                          await cond.wait()
 563                      if not stack:
 564                          cond.notify_all()
 565                          return
 566                      name, url, cid, depth = stack.pop(0)
 567                      nid = cid.split("_")[-1]
 568                      if nid in seen:
 569                          if seen[nid] != cid:
 570                              self.ck.event("alias_node_skipped", {"category": cat.name, "node": name, "path": cid,
 571                                                                   "planned_as": seen[nid]})
 572                          continue
 573                      seen[nid] = cid
 574                      active += 1
 575                  try:
 576                      kids = await visit(name, url, cid, depth)
 577                      async with cond:
 578                          stack[0:0] = kids
 579                  finally:
 580                      async with cond:
 581                          active -= 1
 582                          cond.notify_all()
 583  
 584          await asyncio.gather(*(worker() for _ in range(int(self.crawl.get("planning_concurrency", 4)))))
 585          self.ck.set(f"discovered:{cat.name}", True)
 586          rows = [l for l in self.ck.listings() if l["category"] == cat.name]
 587          log.info("category_discovered", extra={"category": cat.name, "listings": len(rows),
 588                                                 "walmart_total_planned": sum(l["reported_total"] or 0 for l in rows)})
 589  
 590      async def _plan_node(self, category, sub, url, cid, depth, depth_max, excluded, skip) -> dict | None:
 591          max_pages = int(self.crawl.get("max_pages_per_listing", 25))
 592          max_kids = int(self.crawl.get("max_split_children", 30))
 593          try:
 594              page_url = with_params(url, **self._listing_params())
 595              res = await self.fetcher.get(page_url)
 596              res, lp = await self._get_listing(page_url, res)
 597          except ShapeError as exc:
 598              kids = self._landing_children(res, cid, excluded, skip)
 599              if kids and depth < depth_max:
 600                  log.info("landing_page_split", extra={"category": category, "node": sub, "children": len(kids)})
 601                  return {"kids": kids, "landing": True}
 602              if depth:
 603                  self.ck.event("landing_page_without_listing", {"category": category, "node": sub, "url": url})
 604                  return {"landing_without_listing": True}
 605              self.ck.add_listing(category, sub, url)
 606              self.ck.fail_listing(category, url, f"discovery failed: {exc}")
 607              return None
 608          except (FetchError, ChallengeError) as exc:
 609              self.ck.add_listing(category, sub, url)
 610              self.ck.fail_listing(category, url, f"discovery failed: {exc}")
 611              log.error("discovery_failed", extra={"category": category, "url": url, "error": str(exc)})
 612              return None
 613          if not self._check_context(lp.location, url):
 614              self.ck.add_listing(category, sub, url)
 615              self.ck.fail_listing(category, url, f"page 1: selected store context {lp.location}")
 616              return None
 617          total = lp.total_count
 618          raw = [(n, k) for n, k in lp.children if k.startswith(cid + "_")]
 619          if self._fits(lp, max_pages, has_kids=bool(raw)):
 620              self._plan_leaf(category, sub, url, lp, res, max_pages)
 621              return {"leaf": True, "walmart_total": total}
 622          kids = [(n, k) for n, k in raw if n.lower() not in excluded and not any(p.search(n) for p in skip)]
 623          ids = {k for _n, k in kids}
 624          kids = [(n, k) for n, k in kids if not any(k != o and k.startswith(o + "_") for o in ids)]
 625          has_excluded = len(kids) < len(raw)  # an excluded subcategory must never be pulled in by a price band
 626          if kids and depth < depth_max and (len(kids) <= max_kids or has_excluded):
 627              log.info("category_split", extra={"category": category, "node": sub, "walmart_total": total,
 628                                                "children": len(kids)})
 629              return {"kids": kids, "walmart_total": total}
 630          reason = "no_subcategories" if not kids else "max_depth" if depth >= depth_max else "merchandising_fan_out"
 631          self.ck.event("price_partition", {"category": category, "sub_category": sub, "url": url,
 632                                            "walmart_total": total, "reason": reason, "children": len(kids)})
 633          n, failed = await self._source_bands(category, sub, url, total, max_pages)
 634          return None if failed else {"bands": n, "walmart_total": total, "reason": reason}

 636      async def _get_listing(self, page_url: str, res=None):
 637          """Fetch + parse one listing page. A page that reports products but serves none is fetched once more
 638          before it is accepted as Walmart's own answer (guards against a transient empty response)."""
 639          res = res or await self.fetcher.get(page_url)
 640          lp = parse_listing(res.html, res.final_url)
 641          if not lp.items and lp.total_count:
 642              res = await self.fetcher.get(page_url)
 643              lp = parse_listing(res.html, res.final_url)
 644              self.ck.event("empty_page_refetched", {"url": page_url, "walmart_total": lp.total_count,
 645                                                     "items_after_refetch": len(lp.items), "final_url": res.final_url})
 646          return res, lp

 648      def _fits(self, lp, max_pages: int, has_kids: bool = False) -> bool:
 649          """All of a listing's products are reachable: Walmart exposes fewer pages than the window, its total fits the
 650          window, or - for a node without subcategories - its first page is not even full: then Walmart's total is not
 651          backed by served products (e.g. a search-fallback count) and the listing is paginated until Walmart stops
 652          serving products. A node WITH subcategories is always split (a category root may show a short first page)."""
 653          if lp.max_page < max_pages or lp.total_count <= self._window():
 654              return True
 655          return not has_kids and len(lp.items) < int(self.crawl.get("full_page_min_items", 20))

 657      def _plan_leaf(self, category, sub, url, lp, res, max_pages) -> None:
 658          self.ck.add_listing(category, sub, url)
 659          if self.ck.listing_row(category, url)["next_page"] == 1:
 660              self._process_page(category, sub, url, 1, lp, res, max_pages)

 662      async def _source_bands(self, category, sub, url, node_total, max_pages) -> tuple[int, int]:
 663          """Whole-dollar price bands (Walmart ignores fractional min/max) of a node, each opened store-pinned and
 664          Walmart-sold, split until its Walmart total fits the page window. A band that does not shrink although
 665          narrower means Walmart ignores the price filter there: recorded as an unresolved route (blocks publication)."""
 666          edges = [float(x) for x in self.crawl.get("price_band_edges", [0, 3, 5, 10, 20, 50, 100, 100000])]
 667          full = (edges[0], edges[-1])
 668          bands = list(zip(edges[:-1], edges[1:]))
 669          n = failed = 0
 670          while bands:
 671              lo, hi = bands.pop(0)
 672              band_url = with_params(url, min_price=f"{lo:g}", max_price=f"{hi:g}")
 673              try:
 674                  res, lp = await self._get_listing(with_params(band_url, **self._listing_params()))
 675              except (FetchError, ShapeError, ChallengeError) as exc:
 676                  self.ck.add_listing(category, sub, band_url)
 677                  self.ck.fail_listing(category, band_url, f"price band failed: {exc}")
 678                  failed += 1
 679                  continue
 680              if not self._check_context(lp.location, band_url):
 681                  self.ck.add_listing(category, sub, band_url)
 682                  self.ck.fail_listing(category, band_url, f"page 1: selected store context {lp.location}")
 683                  failed += 1
 684                  continue
 685              total = lp.total_count
 686              fits = self._fits(lp, max_pages)
 687              if not fits and (lo, hi) != full and total >= 0.95 * node_total:
 688                  self.ck.event("price_filter_ignored", {"category": category, "sub_category": sub, "url": band_url,
 689                                                         "band": [lo, hi], "walmart_total": total, "node_total": node_total})
 690                  self.ck.add_listing(category, sub, band_url)
 691                  self.ck.fail_listing(category, band_url, f"unresolved route: Walmart ignores the price filter "
 692                                                           f"({total} of {node_total} results in band {lo:g}-{hi:g})")
 693                  return n, failed
 694              if not fits and hi - lo > 1:
 695                  mid = int((lo + hi) // 2) if hi < 100000 else int(lo * 2 + 50)
 696                  bands[0:0] = [(lo, mid), (mid, hi)]
 697                  continue
 698              self.ck.event("price_band", {"category": category, "sub_category": sub, "band": [lo, hi], "walmart_total": total})
 699              self._plan_leaf(category, sub, band_url, lp, res, max_pages)  # a $1 band still too large ends capped
 700              n += 1
 701          log.info("price_partitioned", extra={"category": category, "sub_category": sub, "bands": n})
 702          return n, failed

 704      async def _price_partitions(self, category: str, sub: str, url: str) -> list[tuple[str, str, int]]:
 705          """Split an oversize store assortment into whole-dollar price bands (Walmart honours whole-dollar
 706          min_price / max_price only) until each band's store assortment fits the store window."""
 707          edges = [float(x) for x in self.crawl.get("price_band_edges", [0, 3, 5, 10, 20, 50, 100, 100000])]
 708          bands = list(zip(edges[:-1], edges[1:]))
 709          out: list[tuple[str, str, int]] = []
 710          while bands:
 711              lo, hi = bands.pop(0)
 712              band_url = with_params(url, min_price=f"{lo:g}", max_price=f"{hi:g}")
 713              total = await self._store_count(band_url)
 714              if total is None:
 715                  self.ck.add_listing(category, sub, band_url)
 716                  self.ck.fail_listing(category, band_url, "price band count failed")
 717                  continue
 718              if total > self._store_window() and hi - lo > 1:
 719                  mid = int((lo + hi) // 2) if hi < 100000 else int(lo * 2 + 50)
 720                  bands[0:0] = [(lo, mid), (mid, hi)]
 721                  continue
 722              self.ck.event("price_band", {"category": category, "sub_category": sub, "band": [lo, hi], "store_assortment": total})
 723              if total > self._store_window():
 724                  log.warning("price_band_still_too_large", extra={"category": category, "sub_category": sub,
 725                                                                   "band": [lo, hi], "store_assortment": total})
 726              out.append((sub, band_url, total))
 727          log.info("price_partitioned", extra={"category": category, "sub_category": sub, "bands": len(out)})
 728          return out

 730      @staticmethod
 731      def _landing_children(res, cid: str, excluded: set, skip: list) -> list[tuple[str, str]]:
 732          try:
 733              data = next_data(res.html, res.final_url)
 734          except (ShapeError, ChallengeError):
 735              return []
 736          pairs = [(n, category_id_of(u)) for n, u in discover_links(data, res.html)] + department_children(data)
 737          kids, seen = [], set()
 738          for n, k in pairs:

  59  def walk(node: Any) -> Iterator[Any]:
  60      """Breadth-first walk that preserves document order."""
  61      queue = deque([node])
  62      while queue:
  63          cur = queue.popleft()
  64          yield cur
  65          if isinstance(cur, dict):
  66              queue.extend(cur.values())
  67          elif isinstance(cur, list):
  68              queue.extend(cur)

  72      for p in path:
  73          if isinstance(node, dict) and p in node:
  74              node = node[p]
  75          elif isinstance(node, list) and isinstance(p, int) and -len(node) <= p < len(node):
  76              node = node[p]
  77          else:
  78              return default

 107  def store_context_excerpt(data: Any, limit: int = 40) -> list[dict]:
 108      """JSON paths + values of every store / pickup / postal-code field in a page (evidence of context)."""
 109      out: list[dict] = []
 110      queue = deque([("$", data)])
 111      while queue and len(out) < limit:
 112          path, node = queue.popleft()
 113          if isinstance(node, dict):

 115                  p = f"{path}.{k}"
 116                  if k in _CONTEXT_KEYS and isinstance(v, (str, int, list)) and v not in ("", [], None):
 117                      out.append({"path": p, "value": v if not isinstance(v, list) else v[:10]})
 118                  if isinstance(v, (dict, list)):

 120          elif isinstance(node, list):

 375  def discover_links(data: Any, html: str = "") -> list[tuple[str, str]]:
 376      """Collect (name, url) pairs for /browse/ category links on a category or listing page."""
 377      seen, out = set(), []
 378      for node in walk(data):
 379          if not isinstance(node, dict):
 380              continue
 381          url = node.get("url")
 382          if not isinstance(url, str) and isinstance(node.get("clickThrough"), dict):
 383              url = node["clickThrough"].get("value")
 384          name =node.get("name") or node.get("title") or node.get("text") or node.get("displayName")
 385          if isinstance(url, str) and "/browse/" in url and isinstance(name, str):
 386              path = url.split("?")[0]
 387              if path not in seen:
 388                  seen.add(path)
 389                  out.append((clean_text(name), urljoin(BASE, path)))
 390      for href, _id, label in _BROWSE_HREF.findall(html or ""):
 391          if href not in seen:
 392              seen.add(href)
 393              out.append((clean_text(re.sub("<[^>]+>", " ", label)), urljoin(BASE, href)))

  26  def validate_record_contract(records: list[dict]) -> list[str]:
  27      """Validate identity and store-offer field shapes without recalculating PPM."""
  28      errors: list[str] = []
  29      for index, record in enumerate(records):
  30          prefix = f"record {index}"
  31          upc = record.get("upc", "")
  32          if upc and not _GTIN_RE.fullmatch(str(upc)):
  33              errors.append(f"{prefix}: invalid UPC/GTIN")
  34          for field in ("regular_price", "sale_price"):
  35              value = record.get(field, "")
  36              if value and not (_PRICE_RE.fullmatch(str(value)) and float(value) > 0):
  37                  errors.append(f"{prefix}: invalid {field}")
  38          for field in WALMART_EMPTY_OFFER_FIELDS:
  39              if record.get(field, "") != "":
  40                  errors.append(f"{prefix}: {field} must be empty for Walmart")
  41          ppm = record.get("ppm", "")
  42          if ppm and not _PPM_RE.match(str(ppm)):
  43              errors.append(f"{prefix}: invalid PPM")
  44          if record.get("sale_price") and not record.get("regular_price"):
  45              errors.append(f"{prefix}: sale price has no regular-price offer")
  46      return errors

   1  """Builds the delivered source package.
   2  
   3    python tools/package_source.py
   4  
   5  Creates dist/walmart-scraper-<version>-src.zip containing the complete, unencrypted source,
   6  config, deployment files, docs and tests, plus MANIFEST.txt (SHA-256 of every file) and
   7  SOURCE_FINGERPRINT.txt. The fingerprint is the same value the scraper writes into every
   8  run report and log (report.json → scraper.source_sha256), which ties each dataset to the
   9  exact code that produced it. Secrets (.env), outputs and caches are never included.
  10  """
  11  from __future__ import annotations
  12  
  13  import hashlib
  14  import sys
  15  import zipfile
  16  from pathlib import Path
  17  
  18  ROOT = Path(__file__).resolve().parents[1]
  19  sys.path.insert(0, str(ROOT / "src"))
  20  
  21  from wmscraper import __version__  # noqa: E402
  22  from wmscraper.version import fingerprint, source_files  # noqa: E402
  23  
  24  EXTRA = ["README.md", ".env.example", ".gitignore", ".dockerignore", "docs",
  25           "walmart-500-demo.json"]  # offline fixture read by tests/test_v112_offline.py
  26  
  27  
  28  def main() -> Path:
  29      files = source_files(ROOT)
  30      for item in EXTRA:
  31          p = ROOT / item
  32          files += [p] if p.is_file() else [f for f in p.rglob("*") if f.is_file()]
  33      files = sorted(set(files), key=lambda f: f.relative_to(ROOT).as_posix())
  34      assert not any(f.name == ".env" for f in files), "secrets must never be packaged"
  35      fp = fingerprint(ROOT)
  36      dist = ROOT / "dist"
  37      dist.mkdir(exist_ok=True)
  38      out = dist / f"walmart-scraper-{__version__}-src.zip"
  39      manifest = [f"{hashlib.sha256(f.read_bytes()).hexdigest()}  {f.relative_to(ROOT).as_posix()}" for f in files]
  40      with zipfile.ZipFile(out, "w", zipfile.ZIP_DEFLATED) as z:
  41          for f in files:
  42              z.write(f, f"walmart-scraper/{f.relative_to(ROOT).as_posix()}")
  43          z.writestr("walmart-scraper/MANIFEST.txt", "\n".join(manifest) + "\n")
  44          z.writestr("walmart-scraper/SOURCE_FINGERPRINT.txt", f"version {__version__}\nsource_sha256 {fp}\n")
  45      zip_sha = hashlib.sha256(out.read_bytes()).hexdigest()
  46      (dist / "SHA256SUMS.txt").write_text(f"{zip_sha}  {out.name}\n", encoding="utf-8")
  47      print(f"{out}\n  files: {len(files)}\n  source_sha256: {fp}\n  zip_sha256: {zip_sha}")
  48      return out
  49  
  50  
  51  if __name__ == "__main__":
  52      main()

  42  def sh(args, log: Path, timeout=None) -> int:
  43      with log.open("w", encoding="utf-8") as fh:
  44          fh.write(f"$ {' '.join(map(str, args))}\n# started {datetime.now(timezone.utc).isoformat(timespec='seconds')}\n")
  45          fh.flush()
  46          p = subprocess.run(args, cwd=ROOT, env=ENV, stdout=fh, stderr=subprocess.STDOUT, timeout=timeout)
  47          fh.write(f"\n# finished {datetime.now(timezone.utc).isoformat(timespec='seconds')} exit code {p.returncode}\n")
  48      return p.returncode

  51  def committed_pages(ckpt: Path) -> int:
  52      try:
  53          return sqlite3.connect(ckpt).execute("SELECT COALESCE(SUM(next_page - 1), 0) FROM listings").fetchone()[0]
  54      except sqlite3.Error:
  55          return 0

  58  def export_traces(log_file: Path, out_csv: Path) -> None:
  59      """Per product: request URL -> final URL -> returned item id / name / UPC -> decision (from the JSON log)."""
  60      if not log_file.exists():
  61          return
  62      with out_csv.open("w", newline="", encoding="utf-8") as fh:
  63          w = csv.writer(fh)
  64          w.writerow(["item_id", "match", "request_url", "final_url", "response_item_id", "response_name",
  65                      "response_upc", "output_title", "decision", "attempts"])
  66          for line in log_file.read_text(encoding="utf-8").splitlines():
  67              try:
  68                  e = json.loads(line)
  69              except ValueError:
  70                  continue
  71              if e.get("event") == "product_identity":
  72                  a = e.get("attempts") or [["", "", ""]]
  73                  w.writerow([e["item_id"], e["match"], a[0][0], a[-1][1], e["response_item_id"], e["response_name"],
  74                              e["response_upc"], e["output_title"], e["decision"], len(a)])

  77  def section_repeat(out: Path, loc, st, first: dict, max_requests: int) -> dict:
  78      """REAL repeat collection: a second fresh run of the same scope, validated against the first published run."""
  79      d = out / "B2-live-repeat-run"
  80      d.mkdir()
  81      cmd = [PY, "-m", "wmscraper", "run", "--location", loc.key, "--fresh"] + (["--max-requests", str(max_requests)] if max_requests else [])
  82      sh(cmd, d / "01-repeat-run.log")
  83      state = st.get_json(f"{loc.key}/latest/manifest.json") or {}
  84      runs = sorted(p.name for p in st.local_path(f"{loc.key}/runs/x").parent.iterdir() if p.is_dir() and p.name[0].isdigit())
  85      run_id = [r for r in runs if r > first["run_id"]][-1]
  86      run_dir = st.local_path(f"{loc.key}/runs/{run_id}/records.json").parent
  87      rep = json.loads((run_dir / "report.json").read_text(encoding="utf-8"))
  88      shutil.copy(run_dir / "records.json", d / "02-repeat-dataset.json")
  89      shutil.copy(run_dir / "report.json", d / "03-repeat-run-report.json")
  90      if (run_dir / "run.log").exists():
  91          shutil.copy(run_dir / "run.log", d / "04-repeat-structured-log.jsonl")
  92      a = {r["url"].split("?")[0]: r for r in json.loads((out / "B-live-walmart" / "04-dataset.json").read_text(encoding="utf-8"))}
  93      b = {r["url"].split("?")[0]: r for r in json.loads((run_dir / "records.json").read_text(encoding="utf-8"))}
  94      with (d / "05-run1-vs-repeat-by-category.csv").open("w", newline="", encoding="utf-8") as fh:
  95          w = csv.writer(fh)
  96          w.writerow(["category", "run1_records", "repeat_records", "in_both", "price_changed"])
  97          for cat in sorted({r["category"] for r in a.values()} | {r["category"] for r in b.values()}):
  98              ka = {k for k, r in a.items() if r["category"] == cat}
  99              kb = {k for k, r in b.items() if r["category"] == cat}
 100              both = ka & kb
 101              w.writerow([cat, len(ka), len(kb), len(both),
 102                          sum(1 for k in both if a[k]["regular_price"] != b[k]["regular_price"] or a[k]["sale_price"] != b[k]["sale_price"])])
 103      summary = {"run_id": run_id, "records": rep["total_unique_records"], "validation_ok": rep["validation"]["ok"],
 104                 "published": bool(state.get("run_id") == run_id), "compared_with_run": first["run_id"],
 105                 "overlap_with_run1": len(set(a) & set(b)), "yield_check": "real: validated against the previous published run"}
 106      (d / "06-repeat-summary.json").write_text(json.dumps(summary, indent=2), encoding="utf-8")
 107      return summary

 110  def section_blocked(out: Path, loc, st, budget: int = 40) -> dict:
 111      """LIVE failed run on the authoritative version: a fresh run starved of requests cannot finish, must fail
 112      validation and must leave latest/ untouched."""
 113      d = out / "D-live-failed-run-not-published"
 114      d.mkdir()
 115      before = st.get_json(f"{loc.key}/latest/manifest.json")
 116      (d / "01-latest-manifest-BEFORE.json").write_text(json.dumps(before, indent=2), encoding="utf-8")
 117      sh([PY, "-m", "wmscraper", "run", "--location", loc.key, "--fresh", "--max-requests", str(budget)], d / "02-blocked-run.log")
 118      after = st.get_json(f"{loc.key}/latest/manifest.json")
 119      (d / "03-latest-manifest-AFTER.json").write_text(json.dumps(after, indent=2), encoding="utf-8")
 120      runs = sorted(p.name for p in st.local_path(f"{loc.key}/runs/x").parent.iterdir() if p.is_dir() and p.name[0].isdigit())
 121      rep = json.loads(st.local_path(f"{loc.key}/runs/{runs[-1]}/report.json").read_text(encoding="utf-8"))
 122      shutil.copy(st.local_path(f"{loc.key}/runs/{runs[-1]}/report.json"), d / "04-blocked-run-report.json")
 123      # operator closes the deliberately starved run so the schedule does not resume it
 124      state = st.get_json(f"{loc.key}/state.json") or {}
 125      st.put_json(f"{loc.key}/state.json", {**state, "open_run": None, "closed_run": rep["run_id"]})
 126      summary = {"run_id": rep["run_id"], "request_budget": budget, "records_collected": rep["total_unique_records"],
 127                 "validation_ok": rep["validation"]["ok"], "errors": rep["validation"]["errors"][:6],
 128                 "latest_unchanged": before == after, "latest_run_id": (after or {}).get("run_id")}
 129      (d / "05-summary.json").write_text(json.dumps(summary, indent=2), encoding="utf-8")
 130      (d / "README.md").write_text(
 131          "# D. Live failed run - not published\n\nA real, fresh collection on store #100 with a deliberately "
 132          f"insufficient request budget ({budget}). It cannot complete, fails validation and is not published: "
 133          "compare 01 (before) and 03 (after) manifests.\n", encoding="utf-8")
 134      return summary

 137  def section_traversal(out: Path, key: str, st, max_requests: int) -> dict:
 138      """Production traversal (no demo caps) of one category: every listing paginated to exhaustion."""
 139      d = out / "E-production-traversal"
 140      d.mkdir()
 141      cmd = [PY, "-m", "wmscraper", "run", "--location", key, "--fresh"] + (["--max-requests", str(max_requests)] if max_requests else [])
 142      sh(cmd, d / "01-traversal-run.log")
 143      runs = sorted(p.name for p in st.local_path(f"{key}/runs/x").parent.iterdir() if p.is_dir() and p.name[0].isdigit())
 144      run_dir = st.local_path(f"{key}/runs/{runs[-1]}/records.json").parent
 145      rep = json.loads((run_dir / "report.json").read_text(encoding="utf-8"))
 146      shutil.copy(run_dir / "report.json", d / "02-traversal-report.json")
 147      shutil.copy(run_dir / "records.json", d / "03-traversal-dataset.json")
 148      if (run_dir / "run.log").exists():
 149          shutil.copy(run_dir / "run.log", d / "04-traversal-structured-log.jsonl")
 150      with (d / "05-listing-exhaustion.csv").open("w", newline="", encoding="utf-8") as fh:
 151          w = csv.writer(fh)
 152          w.writerow(["category", "sub_category", "listing_url", "status", "pages_fetched", "walmart_max_page",
 153                      "exhausted", "walmart_reported_total", "seen", "kept"])
 154          for l in rep["listings"]:
 155              pages = l["next_page"] - 1
 156              w.writerow([l["category"], l["sub_category"], l["url"], l["status"], pages, l["max_page"],
 157                          pages >= l["max_page"], l.get("reported_total"), l["seen"], l["kept"]])
 158      ex = sum(1 for l in rep["listings"] if l["next_page"] - 1 >= l["max_page"])
 159      summary = {"run_id": rep["run_id"], "listings": len(rep["listings"]), "listings_exhausted": ex,
 160                 "price_partitioned_listings": sum(1 for l in rep["listings"] if "min_price=" in l["url"]),
 161                 "records": rep["total_unique_records"], "validation_ok": rep["validation"]["ok"],
 162                 "requests": rep["fetch_stats"].get("requests"), "runtime_s": rep["runtime_seconds"]}
 163      (d / "06-traversal-summary.json").write_text(json.dumps(summary, indent=2), encoding="utf-8")
 164      (d / "README.md").write_text(
 165          "# E. Production traversal (no demonstration caps)\n\nOne full category of store #100 crawled with production "
 166          "settings: every discovered listing paginated until Walmart's last page; listings beyond the 25-page window are "
 167          "split by subcategory or by price band. Product pages are not fetched in this section (traversal only); UPC "
 168          "enrichment is demonstrated in B.\n", encoding="utf-8")
 169      return summary

 195  def section_live(out: Path, loc, st, kill_after: int, max_requests: int = 0) -> dict:
 196      d = out / "B-live-walmart"
 197      d.mkdir()
 198      sh([PY, "-m", "wmscraper", "verify-store", "--location", loc.key], d / "01-store-verification.log", timeout=900)
 199  
 200      # main live run re-verifies every product page; the repeat run (B2) then shows UPC reuse for known items
 201      cmd = [PY, "-m", "wmscraper", "run", "--location", loc.key]
 202      if max_requests:
 203          cmd += ["--max-requests", str(max_requests)]
 204      started = time.time()
 205      first = cmd + ["--fresh"]  # new run; the restart after the kill uses the plain command and must resume it
 206      previous = (st.get_json(f"{loc.key}/state.json") or {}).get("open_run")
 207      with (d / "02-run-part1-hard-killed.log").open("w", encoding="utf-8") as log1:
 208          log1.write(f"$ {' '.join(first)}\n")
 209          log1.flush()
 210          proc = subprocess.Popen(first, cwd=ROOT, env=ENV, stdout=log1, stderr=subprocess.STDOUT)
 211          run_id, ckpt = None, None
 212          while proc.poll() is None:
 213              state = st.get_json(f"{loc.key}/state.json") or {}
 214              if state.get("open_run") and state["open_run"] != previous:
 215                  run_id = state["open_run"]
 216                  ckpt = st.local_path(f"{loc.key}/runs/{run_id}/checkpoint.sqlite")
 217                  if committed_pages(ckpt) >= kill_after:
 218                      break
 219              time.sleep(2)
 220          killed_at = committed_pages(ckpt) if ckpt else 0
 221          items_at_kill = sqlite3.connect(ckpt).execute("SELECT COUNT(*) FROM items").fetchone()[0] if ckpt else 0
 222          if proc.poll() is None:
 223              proc.kill()
 224              proc.wait()
 225              log1.write(f"\n# HARD KILL pid {proc.pid} (no cleanup) after {killed_at} committed pages, "
 226                         f"{items_at_kill} items saved, run {run_id}\n")
 227      sh(cmd, d / "03-run-resumed.log")
 228      wall = time.time() - started
 229  
 230      run_dir = st.local_path(f"{loc.key}/runs/{run_id}/records.json").parent
 231      rep = json.loads((run_dir / "report.json").read_text(encoding="utf-8"))
 232      records = json.loads((run_dir / "records.json").read_text(encoding="utf-8"))
 233      assert rep["run_id"] == run_id, "resumed run must continue the same run id"
 234      shutil.copy(run_dir / "records.json", d / "04-dataset.json")
 235      shutil.copy(run_dir / "report.json", d / "05-run-report.json")
 236      if (run_dir / "run.log").exists():
 237          shutil.copy(run_dir / "run.log", d / "06-structured-log.jsonl")
 238      if (run_dir / "raw").exists():
 239          shutil.copytree(run_dir / "raw", d / "07-raw-source-responses")
 240      (d / "08-resume-proof.json").write_text(json.dumps({
 241          "run_id": run_id, "killed_after_committed_pages": killed_at, "items_in_checkpoint_at_kill": items_at_kill,
 242          "resumed_same_run_id": rep["run_id"] == run_id, "final_unique_records": rep["total_unique_records"],
 243          "duplicates_in_output": rep["duplicate_records_in_output"],
 244          "events": [e for e in rep["events_tail"] if e["kind"] in ("run_resumed", "run_aborted")]}, indent=2), encoding="utf-8")
 245  
 246      with (d / "09-category-completeness.csv").open("w", newline="", encoding="utf-8") as fh:
 247          w = csv.writer(fh)
 248          w.writerow(["category", "listings", "done", "failed", "walmart_reported", "collected", "kept_records",
 249                      "upc_fill", "ppm_fill", "weight_fill"])
 250          for cat, c in rep["category_completeness"].items():
 251              w.writerow([cat, c["listings"], c["listings_done"], c["listings_failed"], c["walmart_reported"],
 252                          c["collected"], c["kept_records"], c["fill"]["upc"], c["fill"]["ppm"], c["fill"]["weight"]])
 253      with (d / "10-listing-coverage.csv").open("w", newline="", encoding="utf-8") as fh:
 254          w = csv.writer(fh)
 255          w.writerow(["category", "sub_category", "status", "pages", "walmart_reported_total", "seen", "kept", "capped", "error"])
 256          for l in rep["listings"]:
 257              w.writerow([l["category"], l["sub_category"], l["status"], l["next_page"] - 1, l.get("reported_total"),
 258                          l["seen"], l["kept"], l["capped"], l["error"]])
 259  
 260      # SIMULATED check (not a real repeat collection): the live run's data validated against an artificial baseline
 261      prev = dict(rep)
 262      prev["count_by_category"] = {k: v * 3 for k, v in rep["count_by_category"].items()}
 263      prev["_label"] = "SIMULATED BASELINE - artificial previous report (3x counts); not a real collection"
 264      (d / "11-SIMULATED-baseline-report.json").write_text(json.dumps(prev, indent=2), encoding="utf-8")
 265      sh([PY, "-m", "wmscraper", "validate", "--location", loc.key, "--run-id", run_id,
 266          "--previous", str(d / "11-SIMULATED-baseline-report.json")], d / "11-SIMULATED-yield-drop-check.log")
 267      latest = st.get_json(f"{loc.key}/latest/manifest.json")
 268      (d / "12-published-manifest.json").write_text(json.dumps(latest, indent=2), encoding="utf-8")
 269  
 270      # schema + direct-scrape examples
 271      keys_ok = all(tuple(r) == FIELD_ORDER for r in records)
 272      ex = {"schema_field_order": list(FIELD_ORDER), "all_records_match_schema": keys_ok,
 273            "ppm_examples": [r for r in records if r["ppm"]][:5],
 274            "upc_examples": [r for r in records if r["upc"]][:5],
 275            "sale_price_examples": [r for r in records if r["sale_price"]][:5]}  # Walmart offer fields are always ""
 276      (d / "13-schema-and-field-examples.json").write_text(json.dumps(ex, indent=2, ensure_ascii=False), encoding="utf-8")
 277  
 278      export_traces(d / "06-structured-log.jsonl", d / "15-product-identity-trace.csv")
 279      sh([PY, "tools/results_board.py", str(run_dir)], d / "14-results-board.log")
 280      if (ROOT / "entry" / "02-results-board.png").exists():
 281          shutil.copy(ROOT / "entry" / "02-results-board.png", d / "14-results-board.png")
 282      fs = rep["fetch_stats"]
 283      return {"run_id": run_id, "records": rep["total_unique_records"], "runtime_s": rep["runtime_seconds"],
 284              "wall_s": round(wall), "requests": fs.get("requests"), "retries": fs.get("retries"),
 285              "challenges": fs.get("challenges"), "ok": rep["validation"]["ok"], "published": bool(latest and latest["run_id"] == run_id),
 286              "store": rep["store_verification"], "killed_at": killed_at, "schema_ok": keys_ok}

 289  def write_index(out: Path, loc, live: dict | None, azure_dir: Path | None):
 290      v = version_info()
 291      L = ["# Delivery evidence – Walmart location scraper", "",
 292           f"* Version **{v['version']}** · source_sha256 `{v['source_sha256']}` (same value in every report and log)",
 293           f"* Location `{loc.key}` · store #{loc.store_id} · ZIP {loc.zipcode} · {loc.expected_address} · fulfillment {loc.fulfillment}"]
 294      if live:
 295          s = live["store"]
 296          L += [f"* Live run `{live['run_id']}`: {live['records']} unique records · runtime {live['runtime_s']} s · "
 297                f"{live['requests']} provider requests ({live['retries']} retries, {live['challenges']} challenges) · "
 298                f"validation ok: {live['ok']} · published: {live['published']}",
 299                f"* Store verified: {s.get('store_name')} #{s.get('store_id')}, {s.get('store_location')} ({s.get('checked_at')})"]
 300      L += ["", "| Requirement | Evidence |", "|---|---|",
 301            "| Versioned source | `0-source/*-src.zip`, `0-source/SHA256SUMS.txt` |",
 302            "| Unit / simulated tests (not retailer data) | `A-unit-and-simulated/` |",
 303            "| Real dataset ≥ 500 products, exact 22-field schema | `B-live-walmart/04-dataset.json`, `13-schema-and-field-examples.json` |",
 304            "| Store #100 + pickup context in source responses | `B-live-walmart/01-store-verification.log`, `07-raw-source-responses/index.json` (store_context per response) |",
 305            "| Logs | `B-live-walmart/02…`, `03…`, `06-structured-log.jsonl` |",
 306            "| Hard kill + resume | `B-live-walmart/02-run-part1-hard-killed.log` → `03-run-resumed.log`, `08-resume-proof.json` |",
 307            "| Per-category completeness, thin detection | `B-live-walmart/09-category-completeness.csv`, `10-listing-coverage.csv` |",
 308            "| Failed / thin runs cannot replace output | LIVE: `D-live-failed-run-not-published/` (if present) · SIMULATED baseline: `B-live-walmart/11-SIMULATED-*` · offline: `A-…/verification-board.png` |",
 309            "| Real repeat collection | `B2-live-repeat-run/` (second fresh run, validated against the first published run) |",
 310            "| Production traversal beyond 25 pages | `E-production-traversal/05-listing-exhaustion.csv` (exhaustion + price partitioning) |",
 311            "| Product identity / variants | `B-live-walmart/15-product-identity-trace.csv` (request → response id/name/UPC → output) |",
 312            "| PPM, UPC as scraped (Walmart offer fields always empty) | `B-live-walmart/13-schema-and-field-examples.json`, raw responses in `07-…` |",
 313            "| Azure scheduled publishing | `C-azure-deployed/` |" if azure_dir else "| Azure scheduled publishing | (pending) |",
 314            "| Deployment, schedule, operations, cost | source ZIP → `docs/OPERATIONS.md`, `docs/COST.md`, `deploy/azure/` |",
 315            "", "## File checksums", "", "| file | sha256 |", "|---|---|"]
 316      for f in sorted(p for p in out.rglob("*") if p.is_file() and p.name != "DELIVERY.md"):
 317          L.append(f"| {f.relative_to(out).as_posix()} | `{hashlib.sha256(f.read_bytes()).hexdigest()}` |")
 318      (out / "DELIVERY.md").write_text("\n".join(L) + "\n", encoding="utf-8")

 321  def main():
 322      ap = argparse.ArgumentParser()
 323      ap.add_argument("--location", required=True)
 324      ap.add_argument("--kill-after-pages", type=int, default=6)
 325      ap.add_argument("--max-requests", type=int, default=0, help="safety cap on paid requests per run invocation")
 326      ap.add_argument("--azure-dir", help="folder produced by deploy/azure/collect_evidence.sh")
 327      ap.add_argument("--repeat", action="store_true", help="add a real repeat collection (B2)")
 328      ap.add_argument("--blocked", action="store_true", help="add a live run that must fail and not publish (D)")
 329      ap.add_argument("--traversal-location", help="location key for the production traversal section (E)")
 330      ap.add_argument("--out", help="existing evidence folder to extend (e.g. to add the Azure section)")
 331      a = ap.parse_args()
 332      s = load_settings()
 333      loc = s.location(a.location)
 334      st = build_storage(s)
 335      if a.out:
 336          out = Path(a.out)
 337          live = json.loads((out / "live-summary.json").read_text()) if (out / "live-summary.json").exists() else None
 338      else:
 339          out = ROOT / "evidence" / datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%SZ")
 340          out.mkdir(parents=True)
 341          section_source(out)
 342          section_tests(out)
 343          live = section_live(out, loc, st, a.kill_after_pages, a.max_requests)
 344          if a.repeat:
 345              live["repeat"] = section_repeat(out, loc, st, live, a.max_requests)
 346          if a.repeat or a.blocked:
 347              live["blocked"] = section_blocked(out, loc, st)
 348          if a.traversal_location:
 349              live["traversal"] = section_traversal(out, a.traversal_location, st, a.max_requests)
 350          (out / "live-summary.json").write_text(json.dumps(live, indent=2), encoding="utf-8")
 351      azure = None
 352      if a.azure_dir:
 353          azure = out / "C-azure-deployed"
 354          shutil.copytree(a.azure_dir, azure, dirs_exist_ok=True)
 355      write_index(out, loc, live, azure)
 356      print(out / "DELIVERY.md")

 360      main()

  15     Provider: Oxylabs Web API, Walmart endpoint (store_id, fulfillment_type=pickup, delivery_zip per request)
  16               alternatives: Oxylabs Scraper API | Zyte | Playwright (settings.yaml; secrets in Key Vault)
  17     Storage: Blob  <location>/runs/<run_id>/…   <location>/latest/…   checkpoint.sqlite
  18     Logs: JSON lines → Log Analytics (store, category progress, retries, errors, counts, status)

  25  | Correct store | Store page verified (id, ZIP, street address) before collection. Every request is pinned to the store (provider `store_id`, `fulfillment_type=pickup`, `delivery_zip`, listing `stores=<id>`). **Every** listing and product page must report `pageMetadata.location.storeId` = configured store; any mismatch fails the listing and blocks publication. |
  26  | Local pickup / in-store context | Per item: `fulfillmentSummary` must contain `PICKUP` (or in-store) **at the configured store id**; Walmart's secondary "ships instead" stack (`puShipFul`) is excluded. |
  27  | Shipping-only offers | Excluded (`shipping_only_or_not_at_store`). |
  28  | Marketplace prices | Excluded unless the seller is Walmart.com (seller id `F55CDC31…`), checked on listing **and** product page. |
  29  | Sponsored placements | Real local products are kept **once**; repeated sponsored/organic/category appearances are removed by item-id dedup and counted in `duplicates_prevented`. |
  30  | Duplicates | Primary key = Walmart item id across the whole run (pagination overlap, repeated category, sponsored copy). Counted as `duplicates_prevented`. |
  31  | Approximate weight | Walmart `averageWeight` / sold-by-weight flag, title and product size checked for approx / approximately / avg wt / ~ / weight varies → excluded. |
  32  | PPM | Walmart's displayed `UNIT_PRICE` price line copied verbatim (e.g. `10.2 ¢/fl oz`); empty when not shown; never computed. |
  33  | UPC / GUPC | UPC from the product page, never altered. `gupc` always `""`. |
  34  | unit_price | Always `""` (downstream pipeline). |
  35  | Prices | Each listing price field must be one whole price token (`parse/text.listing_price`: typed number > 0, or e.g. `$4.98`; never a number found inside "Only 2 left" / "Save $2"). Rejected values are logged as `price_rejected`. `regular_price` = was-price when on sale, else current; `sale_price` = current when a valid was-price exists. Two-decimal strings. |
  36  | Volume vs digital deals | Walmart: all four fields (`volume_deals_description`, `volume_deals_price`, `digital_deals_description`, `digital_deals_price`) are always `""` (client decision, 1.1.13). Enforced in `models.enforce_empty_fields` on every export path and by validation. Walmart offers are not collected at all: no promotion text is parsed, there is no offer census and no deal examples in `report.json` (`parse/deals.py` removed). |
  37  | Private label | `config/private_label_brands.txt` matched against brand and title start → `store_brand` = `yes` / `no`. |

  39  ## Category discovery
  40  
  41  Each CSV category is opened as a store-scoped listing (`/browse/<category id>`). If Walmart reports more products than
  42  the 25-page window can show, the node is split into the children of its **Departments** navigation facet (depth-first,
  43  direct children only), until every crawled listing fits the window. Exclusions from the CSV notes ("Recipes", "Farm
  44  Animals", "Shop All Foods") and configurable merchandising-shelf patterns are skipped; overlaps are removed by item-id
  45  dedup.
  46  
  47  ## Silent-blocking and completeness detection
  48  
  49  * Bot-challenge markers, block-page redirects, HTTP errors, tiny responses and abnormal latency are detected per request.
  50  * Every parser validates the response **shape**; a changed or partial page raises `ShapeError` instead of returning 0 items.
  51  * A listing that finishes with **zero** items fails validation – a block is never a legitimately empty category.
  52  * Category counts are compared to the last successful run; a drop below `yield_drop_threshold` (60 %) blocks publication.
  53  * Canary items are read at the start and end of the run; failure = run treated as blocked.
  54  * Listings that hit Walmart's 25-page cap are flagged so they can be split further.
  55  
  56  ## Failed runs never replace good data
  57  
  58  1. Every run writes only to `runs/<run_id>/`.
  59  2. `latest/` is updated only after validation passes: stage → keep `previous-*` → swap → manifest (commit point).
  60  3. Blob versioning + soft delete (14 days) give an extra rollback path.
  61  4. An aborted production run keeps its checkpoint and `state.json` points to it; the next run **continues** it.
  62  
  63  ## Checkpointing
  64  
  65  SQLite per run: listing table (next page, status), items table (record per item id), seen table (kept / skipped
  66  reason), meta (store verification, canaries, runtime). Each page's records and the page pointer are written in **one
  67  transaction**, so a crash can neither lose nor duplicate a page. In Azure the database is uploaded to Blob after every
  68  committed page.
