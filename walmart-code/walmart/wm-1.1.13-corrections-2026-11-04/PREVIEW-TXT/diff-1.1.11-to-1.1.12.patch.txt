+def validate_record_contract(records: list[dict]) -> list[str]:
+    """Validate identity and store-offer field shapes without recalculating PPM."""
+    errors: list[str] = []
+    for index, record in enumerate(records):
+        prefix = f"record {index}"
+        upc = record.get("upc", "")
+        if upc and not _GTIN_RE.fullmatch(str(upc)):
+            errors.append(f"{prefix}: invalid UPC/GTIN")
+        for field in ("regular_price", "sale_price", "volume_deals_price", "digital_deals_price"):
+            value = record.get(field, "")
+            if value and not _PRICE_RE.fullmatch(str(value)):
+                errors.append(f"{prefix}: invalid {field}")
+        ppm = record.get("ppm", "")
+        if ppm and not _PPM_RE.match(str(ppm)):
+            errors.append(f"{prefix}: invalid PPM")
+        if record.get("sale_price") and not record.get("regular_price"):
+            errors.append(f"{prefix}: sale price has no regular-price offer")
+    return errors

+def test_walmart_search_route_builds_store_pinned_payload(site):
+    settings = __import__("wmscraper.config", fromlist=["load_settings"]).load_settings(str(site))
+    settings.raw["fetch"]["oxylabs"]["search_payload"] = {
+        "source": "walmart_search",
+        "query": "{query}",
+        "store_id": "{store_id}",
+        "delivery_zip": "{zipcode}",
+        "fulfillment_type": "pickup",
+    }
+    location = settings.location("test-store")
+    payload = classic_payload(settings, location, "https://www.walmart.com/search?q=whole%20milk&page=2")
+    assert payload == {
+        "source": "walmart_search",
+        "query": "whole milk",
+        "store_id": "1234",
+        "delivery_zip": "72712",
+        "fulfillment_type": "pickup",
+    }

+def test_500_product_fixture_is_complete_and_not_published_by_demo():
+    path = Path(__file__).parents[1] / "walmart-500-demo.json"
+    records = json.loads(path.read_text(encoding="utf-8"))
+    assert len(records) == 500
+    assert all(record["gupc"] == "" and record["unit_price"] == "" for record in records)
+    assert all(not validate_record_contract([record]) for record in records)
